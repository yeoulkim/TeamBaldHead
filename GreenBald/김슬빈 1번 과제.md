#include <iostream> //헤더파일
using namespace	std;

const int SIZE = 5; //배열 크기를 5로 고정

int getSum(int arr[]) { //정수형 배열을 함수에 전달
	int sum = 0; //합계를 저장할 변수
	for (int i = 0; i < SIZE; i++) //배열 전체를 반복
		sum += arr[i]; //배열에 요소들을 하나씩 더함
	return sum;
}
//배열의 합계를 구하는 함수

double getAverage(int arr[]) {
	return (double)getSum(arr) / SIZE; //위에서 만든 합계 함수 호출, 정수 결과를 실수로 바꾸기 위한 형변환
}
//배열 평균 계산

static void sortAscending(int arr[]) { //여기 더 해석이 필요...
	for (int i = 0; i < SIZE - 1; i++)
		for (int j = 0; j < SIZE - 1 - i; j++)
			if (arr[j] > arr[j + 1]) //앞의 수가 더 크면
				swap(arr[j], arr[j + 1]); //두 수를 바꾼다
}
//오름차순 정렬 함수

void sortDescending(int arr[]) {
	for (int i = 0; i < SIZE - 1; i++)
		for (int j = 0; j < SIZE - 1 - i; j++)
			if (arr[j] < arr[j + 1]) //오름차순이랑 방향 바뀜 (큰 수가 앞으로 오게)
				swap(arr[j], arr[j + 1]);
}
//내림차순 정렬 함수

void printArray(int arr[]) {
	for (int i = 0; i < SIZE; i++)
		cout << arr[i] << " ";
	cout << endl;
}
//배열 출력 함수


int main() { //프로그램이 시작되는 부분
	int numbers[SIZE]{}; //숫자 5개를 지정할 정수형 배열 선언

	cout << "숫자 5개 입력";
	for (int i = 0; i < SIZE; i++)
		cin >> numbers[i]; //한 개씩 입력을 받아 배열에 저장

	cout << "합계: " << getSum(numbers) << endl;
	cout << "평균: " << getAverage(numbers) << endl;
	//getSum과 getAverage 함수 호출하여 결과 출력

	int choice;
	cout << "정렬 방법 (1: 오름차순, 2: 내림차순): ";
	cin >> choice;
	//사용자에게 정렬 방법을 선택하게 함

	if (choice == 1)
		sortAscending(numbers);
	else if (choice == 2)
		sortDescending(numbers);
	//입력값에 따라 오름차순 또는 내림차순 정렬 수행

	cout << "정렬 결과: ";
	printArray(numbers);

	return 0;

}
